    ```python
    import json, sys, time
    from datetime import datetime, UTC
    from pathlib import Path

    _KNOWN_SOURCES = ("dart", "krx", "news", "macro", "kind")
    _DEFAULT_ALL = ("krx", "news", "macro", "kind")

    def _dispatch():
        """Lazy-import collectors so a missing dep only breaks its own source."""
        from collectors.krx   import collect_krx
        from collectors.news  import collect_news
        from collectors.macro import collect_macro
        from collectors.kind  import collect_kind
        from collectors.dart  import collect_dart
        return {"dart": collect_dart, "krx": collect_krx, "news": collect_news,
                "macro": collect_macro, "kind": collect_kind}

    def _engine():
        from db.engine import get_engine
        return get_engine()

    def cmd_collect_krx(args) -> int:
        stats = _dispatch()["krx"](vault_root=Path(args.vault_root), engine=_engine(), since=args.since)
        print(json.dumps(stats, ensure_ascii=False, default=str))
        return 0 if not stats.get("failed") else 1

    def cmd_collect_news(args) -> int:
        stats = _dispatch()["news"](vault_root=Path(args.vault_root), engine=_engine(),
                                     since=args.since, max_per_feed=args.max_per_feed)
        print(json.dumps(stats, ensure_ascii=False, default=str))
        return 0 if not stats.get("failed") else 1

    def cmd_collect_macro(args) -> int:
        series = args.series.split(",") if args.series else None
        stats = _dispatch()["macro"](vault_root=Path(args.vault_root), engine=_engine(), series=series)
        print(json.dumps(stats, ensure_ascii=False, default=str))
        return 0 if not stats.get("failed") else 1

    def cmd_collect_kind(args) -> int:
        stats = _dispatch()["kind"](vault_root=Path(args.vault_root), engine=_engine(), since=args.since)
        print(json.dumps(stats, ensure_ascii=False, default=str))
        return 0 if not stats.get("failed") else 1

    def cmd_collect_all(args) -> int:
        requested = [s.strip() for s in args.sources.split(",") if s.strip()]
        unknown = [s for s in requested if s not in _KNOWN_SOURCES]
        if unknown:
            print(f"Unknown sources: {sorted(unknown)}", file=sys.stderr)
            return 2   # D-21 fail-fast
        dispatch = _dispatch()
        vault_root = Path(args.vault_root)
        engine = _engine()
        results: dict = {}
        for src in requested:
            t0 = time.monotonic()
            try:
                kwargs = {"vault_root": vault_root, "engine": engine}
                if src in ("krx", "news", "kind"):
                    kwargs["since"] = args.since
                src_stats = dispatch[src](**kwargs)
                status = "partial" if src_stats.get("failed") else "ok"
                results[src] = {
                    "status": status,
                    "docs_processed": src_stats.get("succeeded", 0),
                    "elapsed_ms": int((time.monotonic() - t0) * 1000),
                    **({"failed_count": len(src_stats["failed"])} if src_stats.get("failed") else {}),
                }
            except Exception as exc:   # D-19 swallow per-source
                results[src] = {
                    "status": "error",
                    "error": str(exc),
                    "elapsed_ms": int((time.monotonic() - t0) * 1000),
                }
        report = {"run_at": datetime.now(UTC).isoformat(), "sources": results}
        print(json.dumps(report, ensure_ascii=False), file=sys.stderr)   # D-20 stderr
        any_bad = any(r["status"] in ("error", "partial") for r in results.values())
        return 1 if any_bad else 0
    ```
