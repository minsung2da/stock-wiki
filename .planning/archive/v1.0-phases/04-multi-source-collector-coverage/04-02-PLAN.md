    ```python
    import re, hashlib
    from datetime import datetime, UTC
    from pathlib import Path
    import pandas as pd
    from shared.content_hash import normalize_body
    from shared.frontmatter import FrontMatter, ProvenanceBlock, write_frontmatter

    _TICKER_RE = re.compile(r"^[0-9]{6}$")
    _DATE_RE = re.compile(r"^\d{4}-\d{2}-\d{2}$")

    def compute_body_hash(body: str) -> str:
        return hashlib.sha256(normalize_body(body).encode("utf-8")).hexdigest()

    def vault_path_for_krx(vault_root: Path, date_iso: str, ticker: str) -> Path:
        if not _TICKER_RE.match(ticker): raise ValueError(f"bad ticker {ticker!r}")
        if not _DATE_RE.match(date_iso): raise ValueError(f"bad date {date_iso!r}")
        return Path(vault_root) / "raw" / "krx" / date_iso / f"{ticker}.md"

    def _render_body(ohlcv: pd.DataFrame, flow: pd.DataFrame, short: pd.DataFrame) -> str:
        # Render three markdown sections using `.to_markdown(index=True)`. `tabulate` is a
        # required dep (added below) so .to_markdown always works — no fallback needed.
        parts = [
            "## OHLCV\n\n", ohlcv.to_markdown(index=True),
            "\n\n## Investor Flow (KRW 순매수)\n\n", flow.to_markdown(index=True),
            "\n\n## Short Balance\n\n", short.to_markdown(index=True),
            "\n",
        ]
        return "".join(parts)

    def write_krx_doc(*, vault_root: Path, ticker: str, date_iso: str, corp_code: str | None,
                      company_name: str | None, ohlcv: pd.DataFrame, flow: pd.DataFrame,
                      short: pd.DataFrame, source_url: str = "https://data.krx.co.kr/") -> tuple[Path, str]:
        path = vault_path_for_krx(vault_root, date_iso, ticker)
        path.parent.mkdir(parents=True, exist_ok=True)
        body = _render_body(ohlcv, flow, short)
        content_hash = compute_body_hash(body)
        fm = FrontMatter(provenance=ProvenanceBlock(
            source="krx", source_id=f"{ticker}_{date_iso}",
            source_url=source_url, date=date_iso,
            fetched_at=datetime.now(UTC), content_hash=content_hash,
            corp_code=corp_code, ticker=ticker, company_name=company_name,
            lang="ko", trust_level="trusted",
        ))
        write_frontmatter(str(path), fm, body)
        return path, content_hash
    ```

    ```python
    """KRX collector — COLL-02. No anthropic/openai imports (CI guarded)."""
    from __future__ import annotations
    import time, logging
    from pathlib import Path
    from typing import Any
    from sqlalchemy.engine import Engine
    from shared.portfolio import Portfolio
    from ingest.heartbeat import record_source_run
    from db.entity import resolve_entity
    from collectors.krx import fetcher, writer

    _log = logging.getLogger(__name__)
    __all__ = ["collect_krx"]

    def _today_iso_krx() -> str:
        # KST trading day; Phase 4 cadence. pykrx handles weekends/holidays via empty DF.
        from datetime import datetime
        from zoneinfo import ZoneInfo
        return datetime.now(ZoneInfo("Asia/Seoul")).date().isoformat()

    def collect_krx(*, vault_root: Path = Path("."), engine: Engine | None = None,
                    since: str | None = None) -> dict[str, Any]:
        start = time.monotonic()
        date_iso = since or _today_iso_krx()
        date_str = date_iso.replace("-", "")
        portfolio = Portfolio.load(vault_root)
        scope = portfolio.scope_tickers()
        stats: dict[str, Any] = {"total": len(scope), "succeeded": 0, "skipped": 0, "failed": []}
        holiday_tickers: list[str] = []
        missing_entities: list[str] = []   # R-03: tickers with no entities row
        for ticker in scope:
            try:
                ohlcv = fetcher.fetch_ohlcv(ticker, date_str)
                if ohlcv.empty:
                    holiday_tickers.append(ticker)
                    stats["skipped"] += 1
                    continue
                flow = fetcher.fetch_trading_value(ticker, date_str)
                short = fetcher.fetch_shorting_balance(ticker, date_str)
                ent = resolve_entity(engine, ticker) if engine is not None else None
                if ent is None:
                    # R-03 Option A: missing entity — do not write; surface via heartbeat.
                    missing_entities.append(ticker)
                    stats["failed"].append({"doc": ticker, "error": "missing_entity"})
                    continue
                corp_code = ent.corp_code
                company_name = ent.canonical_name
                path = writer.vault_path_for_krx(vault_root, date_iso, ticker)
                if path.exists():
                    # Compare hash; skip if identical
                    new_body = writer._render_body(ohlcv, flow, short)
                    new_hash = writer.compute_body_hash(new_body)
                    existing_hash = _read_existing_hash(path)
                    if existing_hash == new_hash:
                        stats["skipped"] += 1
                        continue
                writer.write_krx_doc(vault_root=vault_root, ticker=ticker, date_iso=date_iso,
                                     corp_code=corp_code, company_name=company_name,
                                     ohlcv=ohlcv, flow=flow, short=short)
                stats["succeeded"] += 1
            except Exception as exc:
                _log.exception("krx collect failed for %s", ticker)
                stats["failed"].append({"doc": ticker, "error": str(exc)})
        stats["elapsed_ms"] = int((time.monotonic() - start) * 1000)
        extra: dict | None = None
        extra_merge: dict = {}
        if holiday_tickers:
            extra_merge.update({"skipped_holiday": True, "holiday_tickers": holiday_tickers})
        if missing_entities:
            extra_merge["missing_entity"] = missing_entities   # R-03
        if extra_merge:
            extra = extra_merge
        record_source_run("krx", stats,
                          heartbeat_path=vault_root / "ingested/_status/heartbeat.md",
                          extra=extra)
        return stats

    def _read_existing_hash(path: Path) -> str | None:
        try:
            text = path.read_text(encoding="utf-8")
            # reuse DART pattern — frontmatter between --- fences; extract content_hash
            import re as _re
            m = _re.search(r"content_hash:\s*([0-9a-f]+)", text)
            return m.group(1) if m else None
        except OSError:
            return None
    ```
