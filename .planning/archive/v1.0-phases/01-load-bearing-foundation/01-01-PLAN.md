Create `docker-compose.yml` at repo root with:

```yaml
services:
  postgres:
    image: tensorchord/vchord-suite:pg17-latest
    container_name: stock-postgres
    environment:
      POSTGRES_DB: stockwiki
      POSTGRES_USER: stockwiki
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-stockwiki_dev}
    volumes:
      - pgdata:/var/lib/postgresql/data
      - ./scripts/init-extensions.sql:/docker-entrypoint-initdb.d/init-extensions.sql
    ports:
      - "127.0.0.1:5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U stockwiki"]
      interval: 5s
      timeout: 5s
      retries: 5

volumes:
  pgdata:
```

Use a named volume `pgdata` (not bind mount) to avoid WSL2 permission issues (per Research Pitfall 4 and Open Question 3). Bind `127.0.0.1:5432` only (not `0.0.0.0`) for security. Password uses env var substitution with `stockwiki_dev` as dev default.
