    ```python
    """Phase 8 GAP-05: normalize documents.vault_path absolute → relative.

    Strategy: read INGEST_VAULT_ROOT env var (default repo-root/vault).
    For each row where vault_path startswith vault_root.resolve():
        UPDATE vault_path = relative form
    Rows outside vault_root are left as-is (warning logged).
    """
    import os
    from pathlib import Path
    import sqlalchemy as sa
    from alembic import op

    revision = "0006"
    down_revision = "0005"
    branch_labels = None
    depends_on = None

    def upgrade() -> None:
        # Determine repo_root: env var or pwd parent of alembic.ini
        repo_root = Path(os.environ.get("INGEST_REPO_ROOT", os.getcwd())).resolve()
        vault_root = Path(os.environ.get("INGEST_VAULT_ROOT", repo_root / "vault")).resolve()

        conn = op.get_bind()
        rows = conn.execute(sa.text("SELECT id, vault_path FROM documents")).fetchall()

        updated = 0
        for r in rows:
            vp = r.vault_path
            if not vp:
                continue
            try:
                p = Path(vp)
            except Exception:
                continue
            if not p.is_absolute():
                continue
            try:
                # Try vault_root first, then repo_root (private notes live under repo_root/notes)
                if str(p).startswith(str(vault_root)):
                    rel = str(p.relative_to(vault_root))
                elif str(p).startswith(str(repo_root)):
                    rel = str(p.relative_to(repo_root))
                else:
                    continue  # outside both — leave alone
            except ValueError:
                continue
            conn.execute(
                sa.text("UPDATE documents SET vault_path = :rel WHERE id = :id"),
                {"rel": rel, "id": r.id},
            )
            updated += 1
        # Best-effort logging
        print(f"[0006] normalized {updated} absolute vault_path rows to relative")

    def downgrade() -> None:
        # No-op: cannot reliably reconstruct absolute paths
        pass
    ```
