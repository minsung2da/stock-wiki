```bash
uv run --group graph python - <<'PY'
import graphify, inspect, json
report = {"version": getattr(graphify, "__version__", "unknown")}
modules = ["detect","build","cluster","analyze","report","export","extract","cache"]
for m in modules:
    try:
        mod = __import__(f"graphify.{m}", fromlist=["*"])
        report[m] = sorted([n for n in dir(mod) if not n.startswith("_")])
    except Exception as e:
        report[m] = f"IMPORT_FAILED: {e!r}"
# Probe specific functions named in SKILL.md v4 chain
funcs = [
    ("graphify.detect","detect"),
    ("graphify.build","build_from_json"),
    ("graphify.cluster","cluster"),
    ("graphify.cluster","score_all"),
    ("graphify.analyze","god_nodes"),
    ("graphify.analyze","surprising_connections"),
    ("graphify.analyze","suggest_questions"),
    ("graphify.report","generate"),
    ("graphify.export","to_json"),
    ("graphify.export","to_html"),
]
sigs = {}
for mod_name, fname in funcs:
    try:
        mod = __import__(mod_name, fromlist=[fname])
        fn = getattr(mod, fname)
        sigs[f"{mod_name}.{fname}"] = str(inspect.signature(fn))
    except Exception as e:
        sigs[f"{mod_name}.{fname}"] = f"MISSING: {e!r}"
report["signatures"] = sigs
print(json.dumps(report, indent=2, ensure_ascii=False))
PY
```
