Canonical working SQL (from Plan 01 probe-findings.md + Plan 04 worker SUMMARY):
```sql
SET hnsw.iterative_scan = 'relaxed_order';
WITH dense AS (
    SELECT c.id AS chunk_id,
           ROW_NUMBER() OVER (ORDER BY c.embedding <=> CAST(:qvec AS vector(1024))) AS rk
    FROM chunks c
    JOIN documents d ON d.id = c.document_id
    LEFT JOIN entities e ON e.corp_code = :corp_code
    WHERE (:corp_code IS NULL OR EXISTS (SELECT 1 FROM entities e2 WHERE e2.corp_code = :corp_code))
      AND (:source IS NULL OR d.source = :source)
      AND (:date_from IS NULL OR d.first_seen_at >= :date_from)
      AND (:date_to   IS NULL OR d.first_seen_at <  :date_to)
    ORDER BY c.embedding <=> CAST(:qvec AS vector(1024))
    LIMIT 50
),
sparse AS (
    SELECT c.id AS chunk_id,
           ROW_NUMBER() OVER (
               ORDER BY bm25_catalog.search_bm25query(
                   c.bm25_tokens,
                   bm25_catalog.to_bm25query('ix_chunks_bm25'::regclass, CAST(:qtoks AS int[]))
               ) DESC
           ) AS rk
    FROM chunks c
    JOIN documents d ON d.id = c.document_id
    WHERE (:source IS NULL OR d.source = :source)
      AND (:date_from IS NULL OR d.first_seen_at >= :date_from)
      AND (:date_to   IS NULL OR d.first_seen_at <  :date_to)
    LIMIT 50
)
SELECT COALESCE(dense.chunk_id, sparse.chunk_id) AS chunk_id,
       COALESCE(1.0/(60 + dense.rk), 0) + COALESCE(1.0/(60 + sparse.rk), 0) AS rrf_score
FROM dense FULL OUTER JOIN sparse USING (chunk_id)
ORDER BY rrf_score DESC
LIMIT :top_k;
```
