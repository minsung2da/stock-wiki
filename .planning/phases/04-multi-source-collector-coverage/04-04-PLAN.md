    ```python
    """Alias matching for news articles (D-10 step 4, D-11, D-12).

    R-01 design: pre-load a scoped alias inventory ONCE per collector run (single DB
    query), then scan article title + body for any alias as a substring. This fixes
    recall for Korean particles/punctuation/nicknames that token-regex extraction
    misses. Falls back to resolve_entity_by_alias only when the substring scan
    needs as_of temporal disambiguation on edge-case aliases.
    """
    from __future__ import annotations
    from datetime import date
    from typing import Iterable
    from sqlalchemy import text
    from sqlalchemy.engine import Engine

    class NoAliasesSeededError(RuntimeError):
        """R-09: raised at collect_news startup if entity_aliases has no name rows."""
        pass

    def assert_aliases_seeded(engine: Engine) -> None:
        """R-09 startup guard: entity_aliases must have ≥1 'name'/'eng_name' row."""
        with engine.connect() as conn:
            n = conn.execute(text(
                "SELECT COUNT(*) FROM entity_aliases WHERE kind IN ('name','eng_name')"
            )).scalar() or 0
        if n == 0:
            raise NoAliasesSeededError(
                "No name aliases seeded — run `uv run python -m src.db.seed_name_aliases` first"
            )

    def load_scoped_aliases(
        engine: Engine, tickers: Iterable[str], as_of: date | None = None
    ) -> dict[str, dict]:
        """Return {alias_text: {corp_code, ticker, name}} for entities whose
        current_ticker is in `tickers` (watchlist ∪ holdings per D-12).

        Single SQL round-trip joining entities + entity_aliases (kind IN
        ('name','eng_name')), with temporal filter against `as_of` when supplied.
        Called once at the start of each collect_news run — NOT per-article.
        """
        tickers = list(set(tickers))
        if not tickers:
            return {}
        if as_of is None:
            sql = text("""
                SELECT a.value AS alias, e.corp_code, e.current_ticker, e.canonical_name
                FROM entity_aliases a
                JOIN entities e USING (corp_code)
                WHERE a.kind IN ('name','eng_name')
                  AND a.valid_to IS NULL
                  AND e.current_ticker = ANY(:tks)
            """)
            params = {"tks": tickers}
        else:
            sql = text("""
                SELECT a.value AS alias, e.corp_code, e.current_ticker, e.canonical_name
                FROM entity_aliases a
                JOIN entities e USING (corp_code)
                WHERE a.kind IN ('name','eng_name')
                  AND a.valid_from <= :asof
                  AND (a.valid_to IS NULL OR a.valid_to > :asof)
                  AND e.current_ticker = ANY(:tks)
            """)
            params = {"tks": tickers, "asof": as_of}
        out: dict[str, dict] = {}
        with engine.connect() as conn:
            for row in conn.execute(sql, params):
                # Skip trivially short aliases (≤1 char) to avoid pathological substring hits.
                if len(row.alias) < 2:
                    continue
                out[row.alias] = {
                    "corp_code": row.corp_code,
                    "ticker": row.current_ticker,
                    "name": row.canonical_name,
                }
        return out

    def match_tickers_in_text(text_blob: str, alias_map: dict[str, dict]) -> list[dict]:
        """Scan `text_blob` (title + body concatenation) for substring hits against
        `alias_map`. Longer aliases matched first (prevents "삼성전자" from being
        shadowed by a hypothetical "삼성" alias). Dedup by corp_code.
        """
        if not text_blob or not alias_map:
            return []
        hits: dict[str, dict] = {}
        # Sort aliases by length desc so the longest canonical form wins when overlap exists.
        for alias in sorted(alias_map.keys(), key=len, reverse=True):
            if alias in text_blob:
                entry = alias_map[alias]
                hits.setdefault(entry["corp_code"], entry)
        return list(hits.values())
    ```
    Note: the legacy helper name `match_tickers(body, engine, scope, as_of)` is NOT provided. Callers use `load_scoped_aliases` once, then `match_tickers_in_text(title + "\n" + body, alias_map)` per article.

    ```python
    import hashlib
    import requests
    from requests.exceptions import ConnectionError as ReqConnectionError, ChunkedEncodingError
    from urllib3.exceptions import ProtocolError
    from tenacity import retry, retry_if_exception_type, stop_after_attempt, wait_exponential

    _RETRYABLE_EXC = (ReqConnectionError, ChunkedEncodingError, ProtocolError)
    USER_AGENT = "stock-wiki-collector/0.1 (+https://github.com/YOURORG/stock)"

    def url_hash8(url: str) -> str:
        return hashlib.sha256(url.encode("utf-8")).hexdigest()[:8]

    def _scheme_ok(url: str) -> bool:
        return url.startswith(("http://", "https://"))

    @retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1.0, min=1.0, max=10.0),
           retry=retry_if_exception_type(_RETRYABLE_EXC), reraise=True)
    def fetch_rss_feed(feed_url: str) -> bytes | None:
        """R-08: use plain requests for RSS XML — trafilatura.fetch_url is tuned for
        HTML articles and has been unreliable on feeds (occasional None on valid XML).
        Returns raw bytes so feedparser can detect encoding from the XML declaration.
        """
        if not _scheme_ok(feed_url):
            return None
        resp = requests.get(feed_url, headers={"User-Agent": USER_AGENT}, timeout=15)
        resp.raise_for_status()
        return resp.content

    @retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1.0, min=1.0, max=10.0),
           retry=retry_if_exception_type(_RETRYABLE_EXC), reraise=True)
    def fetch_article_html(url: str) -> str | None:
        """Article HTML retrieval via trafilatura (keeps its fetching heuristics —
        user-agent rotation, http/2 defaults, redirect handling)."""
        if not _scheme_ok(url):
            return None   # SSRF scheme guard (T-04-10)
        import trafilatura
        return trafilatura.fetch_url(url)

    # Back-compat alias removed intentionally: callers must pick the right function.
    ```

    ```python
    import feedparser, trafilatura
    from dataclasses import dataclass
    from datetime import datetime
    from typing import Iterable

    @dataclass(frozen=True)
    class RSSItem:
        url: str
        title: str
        published: datetime | None

    def parse_rss(rss_content: str | bytes) -> list[RSSItem]:
        # feedparser handles either raw bytes (preferred — encoding detection) or str.
        parsed = feedparser.parse(rss_content)
        out = []
        for e in parsed.entries:
            pub = None
            if getattr(e, "published_parsed", None):
                pub = datetime(*e.published_parsed[:6])
            out.append(RSSItem(url=e.link, title=e.title, published=pub))
        return out

    def extract_first_two_paragraphs(html: str) -> str | None:
        text = trafilatura.extract(html, output_format="txt", include_comments=False,
                                    include_tables=False, include_images=False,
                                    favor_precision=True, deduplicate=True)
        if not text: return None
        paragraphs = [p.strip() for p in text.split("\n\n") if p.strip()]
        if not paragraphs: return None
        return "\n\n".join(paragraphs[:2])
    ```

    ```python
    import re, hashlib
    from datetime import datetime, UTC
    from pathlib import Path
    from shared.content_hash import normalize_body
    from shared.frontmatter import FrontMatter, ProvenanceBlock, write_frontmatter

    _OUTLET_RE = re.compile(r"^(hankyung|edaily)$")
    _YYYYMM_RE = re.compile(r"^\d{6}$")
    _HASH8_RE = re.compile(r"^[0-9a-f]{8}$")

    def vault_path_for_news(vault_root: Path, outlet: str, yyyymm: str, url_hash8: str) -> Path:
        if not _OUTLET_RE.match(outlet): raise ValueError(f"bad outlet {outlet!r}")
        if not _YYYYMM_RE.match(yyyymm): raise ValueError(f"bad yyyymm {yyyymm!r}")
        if not _HASH8_RE.match(url_hash8): raise ValueError(f"bad url_hash8 {url_hash8!r}")
        return Path(vault_root) / "raw" / "news" / f"{yyyymm[:4]}-{yyyymm[4:]}" / f"{outlet}_{url_hash8}.md"

    def _assert_two_paragraph_cap(body: str) -> None:
        # Exactly 2 or fewer paragraphs — copyright hard cap (D-13).
        paragraphs = [p for p in body.split("\n\n") if p.strip()]
        if len(paragraphs) > 2:
            raise ValueError(f"news body exceeds 2-paragraph cap ({len(paragraphs)} paragraphs)")

    def write_news_doc(*, vault_root: Path, outlet: str, url: str, url_hash8: str, yyyymm: str,
                       title: str, published_iso: str, tickers: list[dict], body: str) -> tuple[Path, str]:
        _assert_two_paragraph_cap(body)
        path = vault_path_for_news(vault_root, outlet, yyyymm, url_hash8)
        path.parent.mkdir(parents=True, exist_ok=True)
        content_hash = hashlib.sha256(normalize_body(f"{title}\n\n{body}").encode("utf-8")).hexdigest()
        md_body = f"# {title}\n\n{body}\n"
        fm = FrontMatter(provenance=ProvenanceBlock(
            source="news", source_id=url_hash8, source_url=url,
            date=published_iso[:10], fetched_at=datetime.now(UTC),
            content_hash=content_hash,
            corp_code=tickers[0]["corp_code"] if tickers else None,
            ticker=tickers[0]["ticker"] if tickers else None,
            company_name=tickers[0]["name"] if tickers else None,
            lang="ko", trust_level="semi_trusted",
            # News-specific frontmatter (fields added to ProvenanceBlock in Plan 01 Task 3):
            tickers=tickers,
            outlet=outlet,
            license_flag="summary_only",
        ))
        write_frontmatter(str(path), fm, md_body)
        return path, content_hash
    ```

    ```python
        stats: dict[str, Any] = {"total": 0, "succeeded": 0, "skipped": 0, "failed": []}
        for outlet, urls in FEEDS_BY_OUTLET.items():
            for feed_url in urls:
                try:
                    rss_bytes = client.fetch_rss_feed(feed_url)   # R-08: requests, not trafilatura
                    if not rss_bytes: continue
                    items = fetcher.parse_rss(rss_bytes)[:max_per_feed]
                    for item in items:
                        stats["total"] += 1
                        try:
                            url_hash = client.url_hash8(item.url)
                            pub_iso = item.published.isoformat() if item.published else "1970-01-01"
                            yyyymm = pub_iso[:7].replace("-", "")
                            html = client.fetch_article_html(item.url)   # R-08: trafilatura
                            if not html: continue
                            body = fetcher.extract_first_two_paragraphs(html)
                            if not body: continue
                            # R-01: match against title + body using the preloaded alias inventory.
                            tickers = matcher.match_tickers_in_text(
                                f"{item.title}\n{body}", alias_map
                            )
                            if not tickers:
                                stats["skipped"] += 1
                                continue
                            # Idempotency: if file already exists with same content_hash, skip.
                            path = writer.vault_path_for_news(vault_root, outlet, yyyymm, url_hash)
                            if path.exists():
                                import hashlib as _h
                                from shared.content_hash import normalize_body as _nb
                                new_hash = _h.sha256(_nb(f"{item.title}\n\n{body}").encode("utf-8")).hexdigest()
                                existing_hash = _read_existing_hash(path)
                                if existing_hash == new_hash:
                                    stats["skipped"] += 1
                                    continue
                            writer.write_news_doc(
                                vault_root=vault_root, outlet=outlet, url=item.url,
                                url_hash8=url_hash, yyyymm=yyyymm, title=item.title,
                                published_iso=pub_iso, tickers=tickers, body=body,
                            )
                            stats["succeeded"] += 1
                        except Exception as exc:
                            _log.exception("news item failed")
                            stats["failed"].append({"doc": item.url, "error": str(exc)})
                except Exception as exc:
                    _log.exception("news feed failed")
                    stats["failed"].append({"doc": feed_url, "error": str(exc)})
        stats["elapsed_ms"] = int((time.monotonic() - start) * 1000)
        record_source_run("news", stats, heartbeat_path=vault_root / "ingested/_status/heartbeat.md")
        return stats
    ```
