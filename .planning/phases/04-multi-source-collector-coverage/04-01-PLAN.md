    ```python
    from __future__ import annotations
    from pathlib import Path
    import re
    from pydantic import BaseModel, ConfigDict, Field, field_validator
    import yaml

    _TICKER_RE = re.compile(r"^[0-9]{6}$")

    class PortfolioLoadError(RuntimeError):
        pass

    class Holding(BaseModel):
        model_config = ConfigDict(extra="forbid")
        ticker: str
        qty: float = Field(ge=0)
        avg_cost: float = Field(ge=0)
        @field_validator("ticker")
        @classmethod
        def _v(cls, v: str) -> str:
            if not _TICKER_RE.match(v):
                raise ValueError(f"ticker must be 6 ASCII digits, got {v!r}")
            return v

    class Portfolio(BaseModel):
        model_config = ConfigDict(extra="forbid")
        holdings: list[Holding] = []
        watchlist: list[str] = []
        @field_validator("watchlist")
        @classmethod
        def _v(cls, v: list[str]) -> list[str]:
            for t in v:
                if not _TICKER_RE.match(t):
                    raise ValueError(f"watchlist ticker must be 6 ASCII digits, got {t!r}")
            return v

        def scope_tickers(self) -> list[str]:
            return sorted({h.ticker for h in self.holdings} | set(self.watchlist))

        @classmethod
        def load(cls, vault_root: Path) -> "Portfolio":
            p = Path(vault_root) / "notes" / "portfolio.md"
            if not p.exists():
                raise PortfolioLoadError(f"portfolio.md not found at {p}")
            text = p.read_text(encoding="utf-8")
            # Parse YAML frontmatter between leading `---` fences; reuse pattern from src/shared/frontmatter.py.
            if not text.startswith("---\n"):
                raise PortfolioLoadError("portfolio.md missing YAML frontmatter")
            end = text.find("\n---", 4)
            if end < 0:
                raise PortfolioLoadError("portfolio.md frontmatter not terminated")
            data = yaml.safe_load(text[4:end]) or {}
            return cls(**data)
    ```

    ```python
    _NAME_MAX_LEN = 128

    def resolve_entity_by_alias(
        engine: Engine,
        name: str,
        as_of: date | None = None,
    ) -> Entity | None:
        """Resolve Korean/English company name to Entity via entity_aliases (D-11).

        Exact match only (no fuzzy/substring). Filters kind IN ('name','eng_name')
        matching the CHECK constraint in migration 0001. Temporal semantics mirror
        resolve_entity: as_of=None → current (valid_to IS NULL); as_of=<date> →
        half-open interval [valid_from, valid_to).
        """
        if not name or len(name) > _NAME_MAX_LEN:
            return None
        if as_of is None:
            sql = text("""
                SELECT e.corp_code, e.canonical_name, e.current_ticker
                FROM entity_aliases a
                JOIN entities e USING (corp_code)
                WHERE a.kind IN ('name','eng_name')
                  AND a.value = :v
                  AND a.valid_to IS NULL
                LIMIT 1
            """)
            params: dict[str, object] = {"v": name}
        else:
            sql = text("""
                SELECT e.corp_code, e.canonical_name, e.current_ticker
                FROM entity_aliases a
                JOIN entities e USING (corp_code)
                WHERE a.kind IN ('name','eng_name')
                  AND a.value = :v
                  AND a.valid_from <= :asof
                  AND (a.valid_to IS NULL OR a.valid_to > :asof)
                LIMIT 1
            """)
            params = {"v": name, "asof": as_of}
        with engine.connect() as conn:
            row = conn.execute(sql, params).first()
        if row is None:
            return None
        return Entity(corp_code=row.corp_code, canonical_name=row.canonical_name, current_ticker=row.current_ticker)
    ```

    ```python
    """One-shot seeder: insert a 'name' alias for every entity that lacks one.

    Runs before collect_news so alias matching has rows to match against.
    Idempotent: SELECT-then-INSERT per row (no unique constraint; mirrors upsert_entity pattern).
    """
    from __future__ import annotations
    from datetime import date
    from sqlalchemy import text
    from sqlalchemy.engine import Engine

    def seed_name_aliases(engine: Engine) -> int:
        """Insert (corp_code, 'name', canonical_name, today, NULL) for entities
        lacking a current name alias. Returns count inserted."""
        inserted = 0
        with engine.begin() as conn:
            rows = conn.execute(text("SELECT corp_code, canonical_name FROM entities")).all()
            for r in rows:
                existing = conn.execute(
                    text("""
                        SELECT 1 FROM entity_aliases
                        WHERE corp_code = :cc AND kind = 'name'
                          AND value = :v AND valid_to IS NULL
                        LIMIT 1
                    """),
                    {"cc": r.corp_code, "v": r.canonical_name},
                ).first()
                if existing is None:
                    conn.execute(
                        text("""
                            INSERT INTO entity_aliases (corp_code, kind, value, valid_from, valid_to)
                            VALUES (:cc, 'name', :v, :vf, NULL)
                        """),
                        {"cc": r.corp_code, "v": r.canonical_name, "vf": date.today()},
                    )
                    inserted += 1
        return inserted
    ```
