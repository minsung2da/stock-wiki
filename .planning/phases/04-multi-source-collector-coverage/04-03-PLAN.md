    """ECOS + FRED client wrappers. Fail fast on missing keys."""
    import os
    from typing import TYPE_CHECKING

    class CollectorConfigError(RuntimeError):
        pass

    class MacroEmptyResultError(RuntimeError):
        pass

    def ecos_client():
        key = os.environ.get("ECOS_API_KEY")
        if not key:
            raise CollectorConfigError("ECOS_API_KEY missing in environment")
        from PublicDataReader import Ecos
        return Ecos(service_key=key)

    def fred_client():
        key = os.environ.get("FRED_API_KEY")
        if not key:
            raise CollectorConfigError("FRED_API_KEY missing in environment")
        from fredapi import Fred
        return Fred(api_key=key)
    ```

    Create `src/collectors/macro/fetcher.py`:

    import pandas as pd
    from datetime import date, timedelta
    from collectors.macro.client import MacroEmptyResultError

    def fetch_ecos_series(api, series_id: str, cycle: str, item_code: str,
                         days_back: int = 365) -> list[dict]:
        end = date.today()
        start = end - timedelta(days=days_back)
        fmt = "%Y%m%d" if cycle == "D" else "%Y%m"
        df = api.get_statistic_search(
            통계표코드=series_id, 주기=cycle,
            검색시작일자=start.strftime(fmt), 검색종료일자=end.strftime(fmt),
            통계항목코드1=item_code,
        )
        if df is None or len(df) == 0:
            raise MacroEmptyResultError(f"ECOS empty: {series_id}")
        # Return list of {date, value} — date normalized to YYYY-MM-DD
        out = []
        for _, row in df.iterrows():
            t = str(row["TIME"])
            date_iso = f"{t[:4]}-{t[4:6]}-{t[6:8]}" if len(t) == 8 else f"{t[:4]}-{t[4:6]}-01"
            v = row.get("DATA_VALUE")
            if v is None or str(v).strip() == "": continue
            out.append({"date": date_iso, "value": float(v)})
        return out

    def fetch_fred_series(fred, series_id: str, days_back: int = 365) -> list[dict]:
        series = fred.get_series(series_id, observation_start=(date.today()-timedelta(days=days_back)).isoformat())
        return [{"date": idx.date().isoformat(), "value": float(v)}
                for idx, v in series.items() if not pd.isna(v)]
    ```

    Create `src/collectors/macro/writer.py`:
    ```python
    import re, hashlib
    from datetime import datetime, UTC
    from pathlib import Path
    import yaml
    from shared.content_hash import normalize_body
    from shared.frontmatter import FrontMatter, ProvenanceBlock, write_frontmatter, read_frontmatter

    _SERIES_RE = re.compile(r"^[A-Z0-9_\-]{1,32}$")
    _ALLOWED_SOURCES = frozenset({"ecos", "fred"})

    def vault_path_for_macro(vault_root: Path, source: str, series_id: str) -> Path:
        if source not in _ALLOWED_SOURCES: raise ValueError(f"bad source {source!r}")
        if not _SERIES_RE.match(series_id): raise ValueError(f"bad series_id {series_id!r}")
        return Path(vault_root) / "raw" / "macro" / source / f"{series_id}.md"

    def _render_body(label: str, observations: list[dict]) -> str:
        lines = [f"## {label}\n\n| date | value |\n|---|---|"]
        for o in observations:
            lines.append(f"| {o['date']} | {o['value']} |")
        return "\n".join(lines) + "\n"

    def _read_existing_observations(path: Path) -> list[dict]:
        """Read observations from existing frontmatter.provenance.observations.
        Falls back to empty list if file missing or schema mismatch."""
        if not path.exists(): return []
        try:
            fm, _ = read_frontmatter(str(path))
            obs = fm.provenance.observations or []
            return obs if isinstance(obs, list) else []
        except Exception:
            return []

    def merge_observations(
        existing: list[dict], new: list[dict]
    ) -> tuple[list[dict], list[dict]]:
        """R-06: dedup by (date, value); surface ECOS revisions.

        Returns (merged_observations, revisions) where `revisions` is a list of
        {date, old_value, new_value} dicts — empty when every new value either
        duplicates an existing (date,value) pair or introduces a fresh date.
        Same date with different value → new value wins, revision recorded.
        """
        by_date: dict[str, dict] = {o["date"]: o for o in existing}
        revisions: list[dict] = []
        for o in new:
            prev = by_date.get(o["date"])
            if prev is not None and prev.get("value") != o.get("value"):
                revisions.append({
                    "date": o["date"],
                    "old_value": prev.get("value"),
                    "new_value": o.get("value"),
                })
            by_date[o["date"]] = o   # new wins on same date (dedup on (date,value) otherwise)
        merged = sorted(by_date.values(), key=lambda x: x["date"])
        return merged, revisions

    def write_macro_doc(*, vault_root: Path, source: str, series_id: str, label: str,
                        observations: list[dict], source_url: str) -> tuple[Path, str, bool, list[dict]]:
        """Returns (path, content_hash, rewrote, revisions).
        Merges with any existing observations; emits observations in BOTH
        frontmatter (structured, D-07) and body markdown table (human-readable).
        Skips write if hash unchanged. `revisions` (R-06) is the list of
        {date, old_value, new_value} dicts produced by merge_observations —
        empty when no same-date value change occurred."""
        path = vault_path_for_macro(vault_root, source, series_id)
        path.parent.mkdir(parents=True, exist_ok=True)
        existing = _read_existing_observations(path)
        merged, revisions = merge_observations(existing, observations)
        body = _render_body(label, merged)
        content_hash = hashlib.sha256(normalize_body(body).encode("utf-8")).hexdigest()
        # Compare to existing file's hash by re-reading frontmatter content_hash
        existing_hash = None
        if path.exists():
            try:
                fm_prev, _ = read_frontmatter(str(path))
                existing_hash = fm_prev.provenance.content_hash
            except Exception:
                existing_hash = None
        if existing_hash == content_hash:
            return path, content_hash, False, revisions
        fm = FrontMatter(provenance=ProvenanceBlock(
            source=source, source_id=series_id, source_url=source_url,
            date=datetime.now(UTC).date().isoformat(),
            fetched_at=datetime.now(UTC), content_hash=content_hash,
            corp_code=None, ticker=None, company_name=None,
            lang="ko" if source == "ecos" else "en",
            trust_level="trusted",
            observations=merged,   # D-07: structured frontmatter observations
        ))
        write_frontmatter(str(path), fm, body)
        return path, content_hash, True, revisions
    ```

    Create `src/collectors/macro/__init__.py`:
    ```python
    from __future__ import annotations
    import time, logging
    from pathlib import Path
    from typing import Any
    import yaml
    from sqlalchemy.engine import Engine
    from ingest.heartbeat import record_source_run
    from collectors.macro import client, fetcher, writer
    from collectors.macro.client import MacroEmptyResultError

    _log = logging.getLogger(__name__)
    # R-10: Resolve catalog module-relative by default (walk up to repo root), accept override kwarg.
    _REPO_ROOT = Path(__file__).resolve().parents[3]   # src/collectors/macro/__init__.py → repo root
    _CATALOG_PATH = _REPO_ROOT / ".planning" / "macro_series.yaml"
    _ECOS_URL = "https://ecos.bok.or.kr/"
    _FRED_URL = "https://fred.stlouisfed.org/series/"

    def load_catalog(path: Path | None = None) -> dict:
        resolved = path if path is not None else _CATALOG_PATH
        return yaml.safe_load(Path(resolved).read_text(encoding="utf-8")) or {"ecos": [], "fred": []}

    def collect_macro(*, vault_root: Path = Path("."), engine: Engine | None = None,
                      series: list[str] | None = None,
                      catalog_path: Path | None = None) -> dict[str, Any]:
        # R-12: `engine` is unused here; kept for collect_* signature parity with krx/news/kind.
        start = time.monotonic()
        catalog = load_catalog(catalog_path)
        entries = [("ecos", e) for e in catalog.get("ecos", [])] + \
                  [("fred", e) for e in catalog.get("fred", [])]
        if series is not None:
            want = set(series)
            entries = [(s, e) for (s, e) in entries if e["label"] in want]
        stats: dict[str, Any] = {"total": len(entries), "succeeded": 0, "skipped": 0, "failed": []}
        all_revisions: list[dict] = []   # R-06: aggregated across all series
        ecos_api = fred_api = None
        for source, entry in entries:
            try:
                if source == "ecos":
                    ecos_api = ecos_api or client.ecos_client()
                    obs = fetcher.fetch_ecos_series(ecos_api, entry["series_id"], entry.get("cycle", "D"),
                                                   entry.get("item_code", ""))
                    url = _ECOS_URL
                else:
                    fred_api = fred_api or client.fred_client()
                    obs = fetcher.fetch_fred_series(fred_api, entry["series_id"])
                    url = _FRED_URL + entry["series_id"]
                _, _, rewrote, revs = writer.write_macro_doc(
                    vault_root=vault_root, source=source, series_id=entry["series_id"],
                    label=entry["label"], observations=obs, source_url=url,
                )
                for rev in revs:
                    all_revisions.append({"series_id": entry["series_id"], **rev})   # R-06
                if rewrote: stats["succeeded"] += 1
                else: stats["skipped"] += 1
            except MacroEmptyResultError as exc:
                stats["failed"].append({"doc": entry["label"], "error": str(exc)})
            except Exception as exc:
                _log.exception("macro collect failed for %s", entry.get("label"))
                stats["failed"].append({"doc": entry.get("label", "?"), "error": str(exc)})
        stats["elapsed_ms"] = int((time.monotonic() - start) * 1000)
        extra: dict | None = {"revisions": all_revisions} if all_revisions else None
        record_source_run("macro", stats, heartbeat_path=vault_root / "ingested/_status/heartbeat.md", extra=extra)
        return stats
    ```
