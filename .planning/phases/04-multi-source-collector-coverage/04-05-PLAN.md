    ```python
    import time, urllib.robotparser
    import requests
    from collectors.kind.sources import KIND_BASE

    class RobotsDisallowedError(RuntimeError):
        pass

    USER_AGENT = "stock-wiki-collector/0.1 (+https://github.com/YOURORG/stock)"
    MIN_REQUEST_INTERVAL_SEC = 1.0

    _last_request_ts = 0.0

    def check_robots_txt(robots_txt_content: str, path: str = "/disclosureinfo/") -> None:
        rp = urllib.robotparser.RobotFileParser()
        rp.parse(robots_txt_content.splitlines())
        if not rp.can_fetch(USER_AGENT, KIND_BASE + path):
            raise RobotsDisallowedError(f"robots.txt disallows {path}")

    def _throttled_get(url: str, **kwargs) -> requests.Response:
        global _last_request_ts
        elapsed = time.monotonic() - _last_request_ts
        if elapsed < MIN_REQUEST_INTERVAL_SEC:
            time.sleep(MIN_REQUEST_INTERVAL_SEC - elapsed)
        resp = requests.get(url, headers={"User-Agent": USER_AGENT}, timeout=15, **kwargs)
        _last_request_ts = time.monotonic()
        return resp

    def _throttled_post(url: str, data: dict) -> requests.Response:
        global _last_request_ts
        elapsed = time.monotonic() - _last_request_ts
        if elapsed < MIN_REQUEST_INTERVAL_SEC:
            time.sleep(MIN_REQUEST_INTERVAL_SEC - elapsed)
        resp = requests.post(url, data=data, headers={"User-Agent": USER_AGENT}, timeout=15)
        _last_request_ts = time.monotonic()
        return resp
    ```

    ```python
    from collectors.kind.selectors import NFAITH_LISTING_ROW, NFAITH_CELLS, ParseError

    from collectors.kind import client

    def parse_nfaith_page(html: str) -> list[dict]:
        soup = BeautifulSoup(html, "lxml")
        rows = soup.select(NFAITH_LISTING_ROW)
        if rows is None:
            raise ParseError(f"selector miss: {NFAITH_LISTING_ROW}")
        out = []
        for row in rows:
            try:
                rec = {k: row.select_one(sel).get_text(strip=True) for k, sel in NFAITH_CELLS.items()}
            except AttributeError as exc:
                raise ParseError(f"cell selector miss on row: {exc}") from exc
            out.append(rec)
        return out

        resp = client._throttled_post(KRX_MDC_URL, {"bld": KRX_MDC_BLD[bld_key]})
    ```

    ```python
    import re, hashlib
    from datetime import datetime, UTC
    from pathlib import Path
    from shared.content_hash import normalize_body
    from shared.frontmatter import FrontMatter, ProvenanceBlock, write_frontmatter
    from collectors.kind.sources import KindEventType

    _TICKER_RE = re.compile(r"^[0-9]{6}$")
    _EVENT_DATE_RE = re.compile(r"^\d{8}$")
    _ALLOWED_EVENTS = {e.value for e in KindEventType}

    def vault_path_for_kind(vault_root: Path, event_type: str, ticker: str, event_date: str) -> Path:
        if event_type not in _ALLOWED_EVENTS: raise ValueError(f"bad event_type {event_type!r}")
        if not _TICKER_RE.match(ticker): raise ValueError(f"bad ticker {ticker!r}")
        if not _EVENT_DATE_RE.match(event_date): raise ValueError(f"bad event_date {event_date!r}")
        yyyymm = event_date[:6]
        return Path(vault_root) / "raw" / "kind" / f"{yyyymm[:4]}-{yyyymm[4:]}" / \
               f"{event_type}_{ticker}_{event_date}.md"

    def write_kind_event(*, vault_root: Path, event_type: str, ticker: str, event_date: str,
                         corp_code: str | None, company_name: str | None, reason: str,
                         source_url: str, subtype: str | None = None) -> tuple[Path, str, bool]:
        path = vault_path_for_kind(vault_root, event_type, ticker, event_date)
        path.parent.mkdir(parents=True, exist_ok=True)
        body = f"# {event_type.replace('_', ' ').title()} — {company_name or ticker}\n\n" \
               f"- event_date: {event_date}\n- subtype: {subtype or '-'}\n- reason: {reason}\n"
        content_hash = hashlib.sha256(normalize_body(body).encode("utf-8")).hexdigest()
        if path.exists():
            import re as _re
            m = _re.search(r"content_hash:\s*([0-9a-f]+)", path.read_text(encoding="utf-8"))
            if m and m.group(1) == content_hash: return path, content_hash, False
        fm = FrontMatter(provenance=ProvenanceBlock(
            source="kind", source_id=f"{event_type}_{ticker}_{event_date}",
            source_url=source_url, date=f"{event_date[:4]}-{event_date[4:6]}-{event_date[6:8]}",
            fetched_at=datetime.now(UTC), content_hash=content_hash,
            corp_code=corp_code, ticker=ticker, company_name=company_name,
            lang="ko", trust_level="trusted",
        ))
        write_frontmatter(str(path), fm, body)
        return path, content_hash, True
    ```
