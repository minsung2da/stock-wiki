   ```python
   def build_extraction(engine: Engine) -> dict[str, Any]:
       """SQL → graphify build_from_json compatible dict.

       Reads edges + entities + documents and produces:
         nodes: union of {document ids, ticker codes, sector codes, event ids}
         edges: edges table rows with tag column preserved.
       """
       nodes_by_id: dict[str, dict[str, Any]] = {}
       edges_out: list[dict[str, Any]] = []

       with engine.connect() as conn:
           # entities map for label lookup
           entity_by_ticker: dict[str, str] = {}
           ent_rows = conn.execute(sa.text(
               "SELECT corp_code, canonical_name, current_ticker, sector FROM entities"
           )).mappings().all()
           for r in ent_rows:
               if r["current_ticker"]:
                   entity_by_ticker[r["current_ticker"]] = r["canonical_name"]

           # documents map for label/source_file
           doc_meta: dict[str, dict[str, str]] = {}
           doc_rows = conn.execute(sa.text(
               "SELECT id, vault_path, source FROM documents"
           )).mappings().all()
           for r in doc_rows:
               doc_meta[r["id"]] = {"vault_path": r["vault_path"], "source": r["source"]}

           # edges
           edge_rows = conn.execute(sa.text(
               "SELECT src_type, src_id, dst_type, dst_id, edge_type, tag FROM edges"
           )).mappings().all()

       def _ensure_node(ntype: str, nid: str) -> None:
           if nid in nodes_by_id:
               return
           if ntype == "document":
               meta = doc_meta.get(nid, {})
               label = Path(meta.get("vault_path", nid)).stem if meta else nid[:12]
               source_file = meta.get("vault_path", "")
           elif ntype == "ticker":
               label = entity_by_ticker.get(nid, nid)
               source_file = f"db://entities/{nid}"
           elif ntype == "sector":
               label = nid
               source_file = f"db://sectors/{nid}"
           elif ntype == "event":
               label = nid
               source_file = f"db://events/{nid}"
           else:
               label = nid
               source_file = f"db://{ntype}/{nid}"
           nodes_by_id[nid] = {
               "id": nid, "type": ntype, "label": label, "source_file": source_file,
           }

       for r in edge_rows:
           _ensure_node(r["src_type"], r["src_id"])
           _ensure_node(r["dst_type"], r["dst_id"])
           edges_out.append({
               "src": r["src_id"], "dst": r["dst_id"],
               "edge_type": r["edge_type"], "tag": r["tag"], "weight": 1.0,
           })

       return {
           "nodes": list(nodes_by_id.values()),
           "edges": edges_out,
           "input_tokens": 0,
           "output_tokens": 0,
       }
   ```
