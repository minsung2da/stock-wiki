    Public API:
    ```python
    def record_collector_run(
        engine: Engine,
        source: str,                   # 'dart' | 'krx' | 'news' | 'macro' | 'kind'
        stats: dict[str, Any],         # {total, inserted, updated, skipped, failed, elapsed_ms}
        elapsed_ms: int,
        extra: dict[str, Any] | None = None,
    ) -> int | None:
        """INSERT one row into collector_runs. Returns inserted id or None on failure.

        Best-effort: DB errors are logged at WARNING level and swallowed.
        The collect run NEVER fails because of an observability write.

        Pre-conditions:
        - `engine` is a live SQLAlchemy Engine. If None, INSERT is skipped
          and the function returns None (no exception).
        - `stats` and `extra` MUST be JSON-serializable; non-JSONable values
          (e.g., Path objects, datetimes) are coerced via `json.dumps(default=str)`.
        """
    ```

    Implementation outline:
    ```python
    import json
    import logging
    from typing import Any
    from sqlalchemy import text
    from sqlalchemy.engine import Engine

    _log = logging.getLogger(__name__)
    _ALLOWED_SOURCES = frozenset({"dart", "krx", "news", "macro", "kind"})

    def record_collector_run(
        engine: Engine | None, source: str, stats: dict, elapsed_ms: int,
        extra: dict | None = None,
    ) -> int | None:
        if source not in _ALLOWED_SOURCES:
            raise ValueError(f"unknown source: {source!r}")
        if engine is None:
            return None
        try:
            with engine.begin() as conn:
                row_id = conn.execute(
                    text("""
                        INSERT INTO collector_runs (source, elapsed_ms, stats, extra)
                        VALUES (:source, :ms, CAST(:stats AS jsonb),
                                CASE WHEN :extra IS NULL THEN NULL ELSE CAST(:extra AS jsonb) END)
                        RETURNING id
                    """),
                    {
                        "source": source,
                        "ms": int(elapsed_ms),
                        "stats": json.dumps(stats, default=str, ensure_ascii=False),
                        "extra": json.dumps(extra, default=str, ensure_ascii=False) if extra else None,
                    },
                ).scalar()
            return int(row_id)
        except Exception as exc:
            _log.warning(
                "collector_runs INSERT failed (degrade to stderr-only)",
                extra={"source": source, "error": str(exc)},
            )
            return None
    ```

       ```python
       """CI guard: shared.heartbeat must not exist; no collector imports it."""
       from __future__ import annotations
       import ast
       from pathlib import Path

       _REPO_ROOT = Path(__file__).resolve().parents[1]
       _COLLECTOR_DIRS = ["dart", "krx", "news", "macro", "kind"]

       def test_heartbeat_module_not_on_disk():
           assert not (_REPO_ROOT / "src" / "shared" / "heartbeat.py").exists(), \
               "shared/heartbeat.py resurrected — per Phase 1 SC-3 it must remain deleted"

       def test_no_collector_imports_heartbeat():
           offenders = []
           for src in _COLLECTOR_DIRS:
               for py in (_REPO_ROOT / "src" / "collectors" / src).rglob("*.py"):
                   tree = ast.parse(py.read_text(encoding="utf-8"))
                   for node in ast.walk(tree):
                       if isinstance(node, ast.ImportFrom):
                           if node.module and "heartbeat" in node.module:
                               offenders.append(str(py))
                       elif isinstance(node, ast.Import):
                           for alias in node.names:
                               if "heartbeat" in alias.name:
                                   offenders.append(str(py))
           assert not offenders, f"collectors importing heartbeat: {offenders}"

       def test_heartbeat_not_importable():
           import importlib
           try:
               importlib.import_module("shared.heartbeat")
               raise AssertionError("shared.heartbeat is importable; expected ModuleNotFoundError")
           except ModuleNotFoundError:
               pass  # expected
       ```
