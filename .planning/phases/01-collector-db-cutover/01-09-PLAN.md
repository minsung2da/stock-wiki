    """CI fence — Phase 1 SC-1/SC-5/Veto #9. Writer modules must stay deleted."""
    from __future__ import annotations
    import importlib.util
    from pathlib import Path

    _REPO_ROOT = Path(__file__).resolve().parents[1]
    _COLLECTORS = ("dart", "krx", "news", "macro", "kind")

    def test_writer_files_absent():
        offenders = [
            f"src/collectors/{src}/writer.py"
            for src in _COLLECTORS
            if (_REPO_ROOT / "src" / "collectors" / src / "writer.py").exists()
        ]
        assert not offenders, (
            f"vault writer modules resurrected: {offenders}. "
            "Phase 1 v2.0 Veto #9: Markdown vault is dead; collectors INSERT to Postgres."
        )

    def test_writer_modules_not_importable():
        offenders = []
        for src in _COLLECTORS:
            mod = f"collectors.{src}.writer"
            if importlib.util.find_spec(mod) is not None:
                offenders.append(mod)
        assert not offenders, (
            f"writer modules importable: {offenders}. "
            "Even if the .py file is absent, a stub or __init__-shadow could leak."
        )

    def test_no_collector_imports_writer():
        import ast
        offenders = []
        for src in _COLLECTORS:
            init_path = _REPO_ROOT / "src" / "collectors" / src / "__init__.py"
            tree = ast.parse(init_path.read_text(encoding="utf-8"))
            for node in ast.walk(tree):
                if isinstance(node, ast.ImportFrom):
                    if node.module and node.module.endswith(".writer"):
                        offenders.append(str(init_path))
                    elif node.module == f"collectors.{src}" and any(a.name == "writer" for a in node.names):
                        offenders.append(str(init_path))
        assert not offenders, f"collector __init__.py still references writer: {offenders}"
    ```

    """Phase 1 end-to-end smoke test — vertical slice.

    Asserts that the v2.0 cutover is intact:
    - Migration 0006 is head.
    - At least one collector (macro — simplest) INSERTs into its domain table.
    - collector_runs gets a row per run.
    - writer modules are gone.
    - heartbeat module is gone.
    - vault/raw/ is not created.

    Marked 'slow' because it spins the pg_engine fixture and runs end-to-end.
    """
    from __future__ import annotations
    from datetime import date
    from pathlib import Path
    import pytest
    from sqlalchemy import text

    pytestmark = pytest.mark.slow

    def test_phase01_macro_end_to_end(pg_clean, monkeypatch, tmp_path):
        # Stub fetcher to avoid network
        from collectors.macro import fetcher, collect_macro
        monkeypatch.setattr(fetcher, "fetch_ecos_series",
                            lambda api, sid, c, ic: [{"date": date(2026, 1, 2), "value": 100.0}])
        monkeypatch.setattr(fetcher, "fetch_fred_series",
                            lambda api, sid: [{"date": date(2026, 1, 2), "value": 200.0}])
        # Stub client constructors
        from collectors.macro import client
        monkeypatch.setattr(client, "require_env", lambda _: None)
        monkeypatch.setattr(client, "ecos_client", lambda: object())
        monkeypatch.setattr(client, "fred_client", lambda: object())

        # Minimal catalog
        cat = tmp_path / "macro.yaml"
        cat.write_text(
            'ecos:\n  - {label: "AAA", series_id: "AAA", cycle: "D", item_code: ""}\n'
            'fred:\n  - {label: "BBB", series_id: "BBB"}\n',
            encoding="utf-8",
        )

        stats = collect_macro(engine=pg_clean, catalog_path=cat)
        assert stats["inserted"] == 2 and not stats["failed"]

        with pg_clean.begin() as conn:
            ms_count = conn.execute(text("SELECT count(*) FROM macro_series")).scalar()
            cr_count = conn.execute(text("SELECT count(*) FROM collector_runs WHERE source='macro'")).scalar()
        assert ms_count == 2
        assert cr_count == 1

        # vault/raw/ must not exist
        assert not (Path.cwd() / "vault" / "raw").exists()

    def test_phase01_writer_files_absent():
        repo = Path(__file__).resolve().parents[1]
        for src in ("dart","krx","news","macro","kind"):
            assert not (repo / "src" / "collectors" / src / "writer.py").exists(), src

    def test_phase01_heartbeat_module_absent():
        repo = Path(__file__).resolve().parents[1]
        assert not (repo / "src" / "shared" / "heartbeat.py").exists()
    ```
